        local tabBtn = Instance.new("TextButton")
        tabBtn.Name = name .. "_Tab"
        tabBtn.Size = UDim2.new(1, 0, 0, 32)
        tabBtn.BackgroundColor3 = Colors.Element
        tabBtn.BackgroundTransparency = 1
        tabBtn.BorderSizePixel = 0

        tabBtn.TextColor3 = Colors.TextDim

        tabBtn.TextXAlignment = Enum.TextXAlignment.Left

        tabCorner.Parent = tabBtn

        page.Size = UDim2.new(1, -20, 1, -20)
        page.Position = UDim2.new(0, 10, 0, 10)
        page.BackgroundTransparency = 1
        page.BorderSizePixel = 0

        page.ScrollBarImageColor3 = Colors.Accent
        page.CanvasSize = UDim2.new(0, 0, 0, 0)
        page.AutomaticCanvasSize = Enum.AutomaticSize.Y
        page.Visible = false
        page.Parent = pagesFrame

        local pageList = Instance.new("UIListLayout")
        pageList.Padding = UDim.new(0, 8)
        pageList.SortOrder = Enum.SortOrder.LayoutOrder
        pageList.Parent = page

        local function switchTo()
            if currentTab == tabBtn then return end
            if currentTab then
                TweenService:Create(currentTab, TweenInfo.new(0.15), {
                    BackgroundTransparency = 1,
                    TextColor3 = Colors.TextDim
                }):Play()
                pages[currentTab.Name].Visible = false
            end
            currentTab = tabBtn
            TweenService:Create(tabBtn, TweenInfo.new(0.15), {
                BackgroundTransparency = 0,
                TextColor3 = Colors.Text
            }):Play()
            pages[tabBtn.Name].Visible = true
        end

        tabBtn.MouseButton1Click:Connect(switchTo)
        tabBtn.MouseEnter:Connect(function()
            if currentTab ~= tabBtn then
                TweenService:Create(tabBtn, TweenInfo.new(0.15), {BackgroundTransparency = 0.7}):Play()
            end
        end)
        tabBtn.MouseLeave:Connect(function()
            if currentTab ~= tabBtn then
                TweenService:Create(tabBtn, TweenInfo.new(0.15), {BackgroundTransparency = 1}):Play()
            end
        end)

        pages[tabBtn.Name] = page

        if not currentTab then
            switchTo()
        end

        return page
    end

    local function createButton(parent, text, callback)
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, 0, 0, 36)
        btn.BackgroundColor3 = Colors.Element
        btn.BorderSizePixel = 0
        btn.Text = text
        btn.TextColor3 = Colors.Text
        btn.TextSize = 12
        btn.Font = Enum.Font.GothamSemibold
        btn.AutoButtonColor = false
        btn.Parent = parent

        local btnCorner = Instance.new("UICorner")

        local btnStroke = Instance.new("UIStroke")
        btnStroke.Color = Colors.Accent
        btnStroke.Thickness = 1
        btnStroke.Transparency = 1
        btnStroke.Parent = btn

        btn.MouseEnter:Connect(function()
            TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Colors.ElementHover}):Play()
            TweenService:Create(btnStroke, TweenInfo.new(0.15), {Transparency = 0.5}):Play()
        end)

        btn.MouseLeave:Connect(function()
            TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Colors.Element}):Play()
            TweenService:Create(btnStroke, TweenInfo.new(0.15), {Transparency = 1}):Play()
        end)

        btn.MouseButton1Click:Connect(function()
            TweenService:Create(btn, TweenInfo.new(0.1), {BackgroundColor3 = Colors.Accent}):Play()
            task.wait(0.1)
            TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Colors.ElementHover}):Play()
            if callback then callback() end
        end)

        return btn
    end

        local levelText = myLevelInput.Text

        if character then
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            if fakeName ~= "" and humanoid then
                humanoid.DisplayName = fakeName
                for _, d in ipairs(character:GetDescendants()) do
                    if d:IsA("TextLabel") and tonumber(d.Text) == nil then
                        d.Text = fakeName
                    end
                end
            end
            
            if levelText ~= "" then
                local lvlNum = tonumber(levelText)
                if lvlNum then

                    for _, d in ipairs(character:GetDescendants()) do
                        if d:IsA("TextLabel") then
                            local val = tonumber(d.Text)
                            if val ~= nil then
                                d.Text = tostring(lvlNum)

    local function refreshPlayersList()
        for _, child in ipairs(playersList:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end
        
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= localPlayer then
                local pBtn = Instance.new("TextButton")
                pBtn.Size = UDim2.new(1, -12, 0, 28)
                pBtn.BackgroundColor3 = Colors.ElementHover
                pBtn.BorderSizePixel = 0

                pBtn.TextColor3 = Colors.Text
                pBtn.TextSize = 11
                pBtn.Font = Enum.Font.Gotham
                pBtn.TextXAlignment = Enum.TextXAlignment.Left
                pBtn.AutoButtonColor = false
                pBtn.Parent = playersList
                
                local c = Instance.new("UICorner")
                c.CornerRadius = UDim.new(0, 6)
                c.Parent = pBtn
                
                pBtn.MouseEnter:Connect(function()
                    TweenService:Create(pBtn, TweenInfo.new(0.15), {BackgroundColor3 = Colors.Accent}):Play()
                end)
                
                pBtn.MouseLeave:Connect(function()
                    TweenService:Create(pBtn, TweenInfo.new(0.15), {BackgroundColor3 = Colors.ElementHover}):Play()
                end)
                
                pBtn.MouseButton1Click:Connect(function()
                    selectedTargetPlayer = p
                    dropdownBtn.Text = "اللاعب: " .. p.Name .. " ✓"
                    dropdownBtn.TextColor3 = Colors.Success
                    playersList.Visible = false
                    TweenService:Create(playersList, TweenInfo.new(0.2), {Size = UDim2.new(1, 0, 0, 0)}):Play()
                end)
            end
        end
        
        playersList.CanvasSize = UDim2.new(0, 0, 0, playersListLayout.AbsoluteContentSize.Y + 12)
    end

            refreshPlayersList()
