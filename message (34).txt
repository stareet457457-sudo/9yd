local lastSentTime = 0
local lastFishSent = ""
local isSendingWebhook = false

local function sendDiscordNotification(fishName, isTest)
    if WEBHOOK_URL == "" or not string.find(WEBHOOK_URL, "http") then return false end
    local currentTime = tick()

    if not isTest then
        if isSendingWebhook or (fishName == lastFishSent and (currentTime - lastSentTime) < 8) or (currentTime - lastSentTime) < 3 then
            return false
        end
        isSendingWebhook = true
        lastSentTime = currentTime
        lastFishSent = fishName
    end

    local requestFunc = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
    if not requestFunc then
        warn("Executer does not support HTTP requests!")
        if not isTest then isSendingWebhook = false end
        return false
    end

    local data = {
        ["embeds"] = {{
            ["title"] = isTest and "🧪 Webhook Test Success!" or "🎣 Fish Caught!",
            ["description"] = isTest and "الويب هوك شغال 100% يالذيب!" or "Player **" .. LocalPlayer.Name .. "** caught a fish!",
            ["color"] = isTest and 16766720 or 65280,
            ["fields"] = {{["name"] = "Fish Name", ["value"] = "**" .. fishName .. "**", ["inline"] = true}},
            ["footer"] = {["text"] = "MZN Ultimate • " .. os.date("%X")}
        }}
    }

    local success, err = pcall(function()
        requestFunc({
            Url = WEBHOOK_URL,
            Method = "POST",
            Headers = {["Content-Type"] = "application/json"},
            Body = HttpService:JSONEncode(data)
        })
    end)

    if not isTest then
        task.delay(2, function() isSendingWebhook = false end)
    end

    return success
end

-- Détection du poisson hooked
local function getHookedFishName()
    local char = LocalPlayer.Character
    if not char then return nil end

    for _, v in ipairs(char:GetDescendants()) do
        if (v:IsA("StringValue") or v:IsA("ObjectValue")) and fishSettings[tostring(v.Value)] ~= nil then
            return tostring(v.Value)
        end
        for attrName, attrVal in pairs(v:GetAttributes()) do
            if fishSettings[tostring(attrVal)] ~= nil then
                return tostring(attrVal)
            end
        end
    end

    local pGui = LocalPlayer:FindFirstChild("PlayerGui")
    if pGui then
        for _, gui in ipairs(pGui:GetChildren()) do
            local gName = gui.Name:lower()
            if string.find(gName, "fish") or string.find(gName, "reel") or string.find(gName, "hook") or string.find(gName, "minigame") then
                for _, v in ipairs(gui:GetDescendants()) do
                    if v:IsA("TextLabel") and v.Visible then
                        for fName, _ in pairs(fishSettings) do
                            if string.find(v.Text, fName) then
                                return fName
                            end
                        end
                    end
                end
            end
        end
    end

    return nil
end

        if AUTO_DROP_ENABLED then

            pcall(function()
                ItemUtils.iterate_through_items(Data, function(item)
                    if item and item.guid and item.name and fishSettings[item.name] and fishSettings[item.name].drop then
                        table.insert(itemsToDrop, {guid = item.guid, amount = item.amount or 1})
                    end
                end)
            end)

            for _, data in ipairs(itemsToDrop) do
                if not AUTO_DROP_ENABLED then break end
                for i = 1, data.amount do
                    if not AUTO_DROP_ENABLED then break end
                    dropItem(data.guid)
                    task.wait(DROP_DELAY)
                end
            end
        end
    end
end)

local function saveConfig()
    local config = {
        skipTimer = SKIP_TIMER_SECONDS, recastWait = WAIT_BEFORE_RECAST, keybind = currentKeybind.Name,
        worldX = worldTarget and worldTarget.X or nil, worldY = worldTarget and worldTarget.Y or nil, worldZ = worldTarget and worldTarget.Z or nil,
        fSettings = fishSettings, webhook = WEBHOOK_URL, dropDelay = DROP_DELAY, skipDelay = SKIP_DELAY,
        skipBadFish = SKIP_BAD_FISH_ENABLED
    }
    return pcall(function() writefile(CONFIG_FILE, HttpService:JSONEncode(config)) end)
end

local function loadConfig()
    local ok, data = pcall(function() if isfile and isfile(CONFIG_FILE) then return HttpService:JSONDecode(readfile(CONFIG_FILE)) end end)
    if ok and data then
        SKIP_TIMER_SECONDS = data.skipTimer or SKIP_TIMER_SECONDS; WAIT_BEFORE_RECAST = data.recastWait or WAIT_BEFORE_RECAST
        DROP_DELAY = data.dropDelay or DROP_DELAY; SKIP_DELAY = data.skipDelay or SKIP_DELAY
        WEBHOOK_URL = data.webhook or WEBHOOK_URL
        if data.skipBadFish ~= nil then SKIP_BAD_FISH_ENABLED = data.skipBadFish end
        if data.worldX then worldTarget = Vector3.new(data.worldX, data.worldY, data.worldZ) end
        if data.fSettings then
            for k, v in pairs(data.fSettings) do
                if fishSettings[k] then
                    fishSettings[k].drop   = v.drop   ~= nil and v.drop   or fishSettings[k].drop
                    fishSettings[k].notify = v.notify ~= nil and v.notify or fishSettings[k].notify
                    fishSettings[k].keep   = v.keep   ~= nil and v.keep   or fishSettings[k].keep
                end
            end
        end
    end
end

local function castRod()
    local character = LocalPlayer.Character
    if not character then return false end
    if not character:FindFirstChildOfClass("Tool") then
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Two, false, game); task.wait(0.15)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Two, false, game); task.wait(0.4)
        character = LocalPlayer.Character
        if not character or not character:FindFirstChildOfClass("Tool") then return false end
    end
    local inset = game:GetService("GuiService"):GetGuiInset()
    local cx, cy = 960, 540
    if worldTarget then
        local sp, onScreen = Camera:WorldToScreenPoint(worldTarget)
        if onScreen then cx = sp.X; cy = sp.Y + inset.Y end
    end
    VirtualInputManager:SendMouseButtonEvent(cx, cy, 0, true, game, 1); task.wait(0.05); VirtualInputManager:SendMouseButtonEvent(cx, cy, 0, false, game, 1); task.wait(0.15)
    return true
end

-- 🎣 الحلقة الرئيسية (مع Skip Bad Fish + Webhook)
-- ============================================
local function startAutoFishingLoop()
    task.spawn(function()
        while AUTO_FISHING_ENABLED do
            clearFishESP()

            -- 🛑 إذا كانت السنارة مرمية بالماء أساساً (مكتوب Fishing...) لا ترمي مرة أخرى!
            local castSuccess = false
            if not findActiveBobber() then
                for attempt = 1, MAX_CAST_RETRIES do
                    if not AUTO_FISHING_ENABLED then break end
                    if castRod() then castSuccess = true; break end
                    task.wait(0.5)
                end
            else
                castSuccess = true
            end

            if not AUTO_FISHING_ENABLED then break end
            if not castSuccess then fishToggle.Text = "⌛ ما في سنارة..."; task.wait(1); fishToggle.Text = "AUTO FARM: ON"; continue end
            task.wait(0.3)

            local rod = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
            if not rod then task.wait(0.5); continue end

            local snapshot = {}; for _, d in ipairs(rod:GetDescendants()) do snapshot[d] = true end
            local fishCaught = false; local fishObject = nil; local fishName = nil

            local conn = rod.DescendantAdded:Connect(function(newObj)
                if snapshot[newObj] or fishCaught then return end
                local n = newObj.Name; local skip = {Handle=1,Part=1,Attachment=1,Script=1,LocalScript=1,Sound=1,Weld=1,Motor6D=1,SpecialMesh=1,Decal=1,Constraint=1,BillboardGui=1,SurfaceGui=1,RopeConstraint=1,AlignPosition=1,AlignOrientation=1}
                if skip[n] or newObj:IsA("ValueBase") or newObj:IsA("RemoteEvent") or newObj:IsA("BindableEvent") then return end
                fishCaught = true; fishObject = newObj; fishName = n
            end)

            local elapsed = 0; local timeout = math.max(SKIP_TIMER_SECONDS + 5, 30); local lastBobberSeen = tick()
            local detectedFishName = nil

            while not fishCaught and AUTO_FISHING_ENABLED and elapsed < timeout do
                task.wait(0.15); elapsed = elapsed + 0.15

                -- فحص الاسم من الـ UI أو الشخصية
                detectedFishName = getHookedFishName()
                if detectedFishName then fishName = detectedFishName end

                -- 🛑 فحص تصفية الأسماك الخايسة (Skip Bad Fish)
                if detectedFishName and fishSettings[detectedFishName] and SKIP_BAD_FISH_ENABLED then
                    if fishSettings[detectedFishName].keep == false then
                        fishToggle.Text = "⏭️ Skip: " .. detectedFishName
                        -- إلغاء الصيد عبر ضغط رقم 2 مرتين
                        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Two, false, game); task.wait(0.05)
                        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Two, false, game); task.wait(0.15)
                        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Two, false, game); task.wait(0.05)
                        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Two, false, game); task.wait(0.25)
                        break
                    end
                end
                
                if findActiveBobber() or fishCaught then 
                    lastBobberSeen = tick()
                elseif tick() - lastBobberSeen > ANTI_NOFISHING_TIMEOUT then 
                    fishToggle.Text = "🔄 Re-casting..."; break 
                end
            end
            conn:Disconnect()
            if not AUTO_FISHING_ENABLED then break end

            -- 🛑 عند صيد سمكة مقبولة
            local finalFish = fishName or detectedFishName
            if (fishCaught or finalFish) and (not finalFish or not fishSettings[finalFish] or fishSettings[finalFish].keep ~= false) then
                if fishObject then
                    local target = fishObject:IsA("Model") and (fishObject.PrimaryPart or fishObject:FindFirstChildOfClass("BasePart")) or fishObject
                    if target and target:IsA("BasePart") then makeESP(target, finalFish or "Fish") end
                end

                if finalFish and fishSettings[finalFish] and fishSettings[finalFish].notify then
                    task.spawn(sendDiscordNotification, finalFish, false)
                end

                task.wait(SKIP_DELAY)
                clearFishESP()
            end
            fishToggle.Text = "AUTO FARM: ON"
            task.wait(WAIT_BEFORE_RECAST)
        end
        fishToggle.Text = "AUTO FARM: OFF"; fishToggle.BackgroundColor3 = C.white; fishToggle.BackgroundTransparency = 0.88; fishToggle.TextColor3 = C.white
    end)
end
