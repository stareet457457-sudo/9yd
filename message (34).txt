-- ============================================================
--  ZNAP STANDALONE SCRIPT (Red Rectangular Theme)
--  زر مستطيل أحمر بدون إيموجي في أعلى اليمين (PC & Mobile)
-- ============================================================

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")
local LocalPlayer = Players.LocalPlayer

-- الألوان
local COLOR_RED = Color3.fromRGB(255, 45, 65)       -- أحمر نيون
local COLOR_BG  = Color3.fromRGB(20, 10, 12)       -- خلفية داكنة مائلة للأحمر
local COLOR_HOVER = Color3.fromRGB(200, 30, 50)

-- الإعدادات
local ZnapDistance = 10 -- قوة الارتفاع (10 Studs)
local ZnapUnderground = false -- اجعلها true إذا أردت النزول تحت الأرض

-- دالة تنفيذ الـ Znap
local function TriggerZnap()
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local offsetHeight = ZnapUnderground and -ZnapDistance or ZnapDistance
    root.CFrame = root.CFrame + Vector3.new(0, offsetHeight, 0)
end

-- ============================================================
-- إنشاء واجهة الزر (GUI)
-- ============================================================
local TargetGui = (gethui and gethui()) or game:GetService("CoreGui")
if not pcall(function() return TargetGui.Name end) then
    TargetGui = LocalPlayer:WaitForChild("PlayerGui")
end

-- إزالة أي نسخة سابقة
if TargetGui:FindFirstChild("ZnapMobileGui") then
    TargetGui.ZnapMobileGui:Destroy()
end

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "ZnapMobileGui"
ScreenGui.ResetOnSpawn = false
pcall(function()
    if syn and syn.protect_gui then syn.protect_gui(ScreenGui) end
    ScreenGui.Parent = TargetGui
end)

-- الزر المستطيل
local ZnapBtn = Instance.new("TextButton", ScreenGui)
ZnapBtn.Size = UDim2.new(0, 100, 0, 36)
ZnapBtn.Position = UDim2.new(1, -120, 0, 25) -- أعلى اليمين
ZnapBtn.BackgroundColor3 = COLOR_BG
ZnapBtn.BorderSizePixel = 0
ZnapBtn.Text = "ZNAP"
ZnapBtn.Font = Enum.Font.GothamBold
ZnapBtn.TextSize = 13
ZnapBtn.TextColor3 = COLOR_RED
ZnapBtn.Active = true
ZnapBtn.Draggable = true -- قابل للسحب والتحريك على الشاشة
ZnapBtn.ZIndex = 10

-- حواف ناعمة للمستطيل
local UICorner = Instance.new("UICorner", ZnapBtn)
UICorner.CornerRadius = UDim.new(0, 7)

-- إطار أحمر نيون
local UIStroke = Instance.new("UIStroke", ZnapBtn)
UIStroke.Color = COLOR_RED
UIStroke.Thickness = 1.6
UIStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border

-- أنيميشن الضغط (Touch / Click Animation)
local function AnimateClick()
    local shrink = TweenService:Create(ZnapBtn, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 92, 0, 32),
        BackgroundColor3 = COLOR_HOVER,
        TextColor3 = Color3.fromRGB(255, 255, 255)
    })
    local reset = TweenService:Create(ZnapBtn, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 100, 0, 36),
        BackgroundColor3 = COLOR_BG,
        TextColor3 = COLOR_RED
    })
    shrink:Play()
    shrink.Completed:Connect(function()
        reset:Play()
    end)
end

-- تفعيل عبر اللمس / الماوس (للجوال، الآيباد، والكمبيوتر)
ZnapBtn.Activated:Connect(function()
    AnimateClick()
    TriggerZnap()
end)

-- تفعيل عبر الكيبورد (زر Z للـ PC)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.Z then
        AnimateClick()
        TriggerZnap()
    end
end)
