local function UpdateGauges()
    for _, e in ipairs(ValueLabels) do
        if e and e.L and e.L.Parent then
            local val = LiveData[e.K]
            if val ~= nil then
                e.L.Text = tostring(e.F(val))
                if e.K == "rpm" then
                    local r = val / TuningSettings.maxRPM
                    e.L.TextColor3 = r>.9 and Color3.fromRGB(255,50,50)
                        or r>.75 and Color3.fromRGB(255,180,0)
                        or Color3.fromRGB(255,255,255)
                end
            end
        end
    end
end

-- READ CAR DATA
---------------------------------------------------------
local function ReadCarData(seat, dt)
    local rawSpeed = seat.AssemblyLinearVelocity.Magnitude * 3.6
    SmoothedSpeed  = SmoothedSpeed + (rawSpeed - SmoothedSpeed) * math.min(dt*8, 1)
    LiveData.speed = SmoothedSpeed

    -- بعض سيارات A-Chassis تستخدم Seat عادي بدال VehicleSeat
    local thr = 0
    local brk = 0
    if seat:IsA("VehicleSeat") then
        pcall(function()
            thr = seat.ThrottleFloat or 0
            brk = seat.BrakeFloat or 0
        end)
    end
    
    if thr == 0 then
        if UIS:IsKeyDown(Enum.KeyCode.W) or UIS:IsKeyDown(Enum.KeyCode.Up) then thr = 1 end
        if UIS:IsKeyDown(Enum.KeyCode.S) or UIS:IsKeyDown(Enum.KeyCode.Down) then thr = -1 end
    end
    if brk == 0 then
        if UIS:IsKeyDown(Enum.KeyCode.Space) then brk = 1 end
    end
    
    thr = math.abs(thr)
    brk = math.abs(brk)
    LiveData.throttle = thr
    LiveData.brake    = brk

    local spd = SmoothedSpeed
    
    -- محاولة قراءة القير الحقيقي من A-Chassis
    local carModel = seat:FindFirstAncestorWhichIsA("Model") or seat.Parent
    local driveVals = carModel:FindFirstChild("Drive") and carModel.Drive:FindFirstChild("Values")
    local gearVal = driveVals and driveVals:FindFirstChild("Gear")
    
    if gearVal and gearVal:IsA("ValueBase") then
        LiveData.gear = tostring(gearVal.Value)
    else
        local gear = 1
        if    spd > 170 then gear = 6
        elseif spd > 130 then gear = 5
        elseif spd > 95  then gear = 4
        elseif spd > 60  then gear = 3
        elseif spd > 30  then gear = 2 end
        LiveData.gear = tostring(gear)
    end

    local targetRPM = math.min(spd*80/gear + thr*1500 + 600, TuningSettings.maxRPM)
    SmoothedRPM  = SmoothedRPM + (targetRPM - SmoothedRPM) * math.min(dt*6, 1)
    LiveData.rpm = SmoothedRPM

    LiveData.boost = math.max(0, (thr - 0.35) * TuningSettings.targetBoost * 2.2)

    if LiveData.rpm > 2000 then
        LiveData.temp    = math.min(115, LiveData.temp    + dt*0.4)
        LiveData.oilTemp = math.min(125, LiveData.oilTemp + dt*0.2)
    else
        LiveData.temp    = math.max(85,  LiveData.temp    - dt*0.15)
        LiveData.oilTemp = math.max(85,  LiveData.oilTemp - dt*0.1)
    end
    LiveData.oilPress = math.max(2.5, 4.5 - (LiveData.temp-85)*0.012)
    LiveData.fuel     = math.max(0, LiveData.fuel - thr*dt*0.04)
end

-- ✅ FORCE TUNING - يطبق التعديل كل frame (يتغلب على لعبة)
---------------------------------------------------------
local function EnsureForce(root)
    local att = root:FindFirstChild("MaxxECU_Att")
    if not att then
        att = Instance.new("Attachment")
        att.Name = "MaxxECU_Att"
        att.Parent = root
    end

    local vf = root:FindFirstChild("MaxxECU_VF")
    if not vf then
        vf = Instance.new("VectorForce")
        vf.Name = "MaxxECU_VF"
        vf.Attachment0 = att
        vf.RelativeTo = Enum.ActuatorRelativeTo.World
        vf.Force = Vector3.new(0, 0, 0)
        vf.Parent = root
    end
    return vf
end

local function IsGrounded(root)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {root.Parent}
    params.FilterType = Enum.RaycastFilterType.Exclude
    local result = workspace:Raycast(root.Position, root.CFrame.UpVector * -10, params)
    return result ~= nil
end

local function ForceTuneFrame(seat, dt)
    if not TuningActive then
        pcall(function()
            local root = seat and seat.AssemblyRootPart
            local vf = root and root:FindFirstChild("MaxxECU_VF")
            if vf then vf.Force = Vector3.new(0,0,0) end
        end)
        return
    end
    if not seat or not seat.Parent then return end

    if seat:IsA("VehicleSeat") then
        pcall(function()
            seat.MaxSpeed       = TuningSettings.maxSpeed
            seat.TorqueStrength = TuningSettings.engineTorque
            seat.BrakeForce     = TuningSettings.brakeForce
        end)
    end

    pcall(function()
        local root = seat.AssemblyRootPart
        if not root then return end

        local thr = 0
        local brk = 0
        if seat:IsA("VehicleSeat") then
            pcall(function()
                thr = seat.ThrottleFloat or 0
                brk = seat.BrakeFloat or 0
            end)
        end
        
        if thr == 0 then
            if UIS:IsKeyDown(Enum.KeyCode.W) or UIS:IsKeyDown(Enum.KeyCode.Up) then thr = 1 end
            if UIS:IsKeyDown(Enum.KeyCode.S) or UIS:IsKeyDown(Enum.KeyCode.Down) then thr = -1 end
        end
        if brk == 0 then
            if UIS:IsKeyDown(Enum.KeyCode.Space) then brk = 1 end
        end

        local mass = root.AssemblyMass
        local vf = EnsureForce(root)
        
        if (thr == 0 and brk == 0) or not IsGrounded(root) then
            vf.Force = Vector3.new(0, 0, 0)
            return
        end

        -- نحدد اتجاه وجه السيارة، ونجمد محور Y عشان نمنع الطيران 100%
        local look = root.CFrame.LookVector
        local flatLook = Vector3.new(look.X, 0, look.Z)
        if flatLook.Magnitude > 0 then
            flatLook = flatLook.Unit
        else
            flatLook = Vector3.new(0,0,1)
        end

        local currentVel = root.AssemblyLinearVelocity
        local forwardSpeed = currentVel:Dot(flatLook)
        local forceMag = 0 -- تم الإصلاح: تعريف المتغير عشان ما يكرش السكربت
        
        -- قراءة القير الحالي
        local carModel = root.Parent
        local driveVals = carModel:FindFirstChild("Drive") and carModel.Drive:FindFirstChild("Values")
        local gearVal = driveVals and driveVals:FindFirstChild("Gear")
        local currentGear = gearVal and tonumber(gearVal.Value) or tonumber(LiveData.gear) or 1
        
        -- نظام السرعة الذكي لكل قير (توزيع السرعة القصوى على النمر)
        local gearTargetSpeed = 0
        if currentGear > 0 then
            -- القير 1 يعطيك تقريباً السدس، القير 6 يعطيك السرعة كاملة
            local ratio = math.clamp(currentGear / 6, 0.16, 1.0)
            gearTargetSpeed = TuningSettings.maxSpeed * ratio
        elseif currentGear < 0 then
            -- سرعة الريوس محدودة (15% من السرعة القصوى، وبحد أقصى 100 كم/س)
            gearTargetSpeed = math.min(TuningSettings.maxSpeed * 0.15, 100)
        end
        
        local maxForward = (gearTargetSpeed / 3.6) * 3.5
        
        -- تسارع البانزين (صار يوقف تلقائياً إذا وصل حد القير الحالي)
        if thr > 0 and forwardSpeed < maxForward and currentGear ~= 0 then
            forceMag = TuningSettings.engineTorque * mass * thr * 5
        end
        
        -- الفرامل: حرف S (thr < 0) أو Space (brk > 0) تشتغل كفرامل فقط، وما تسرع السيارة بالريوس أبداً!
        local brakeInput = brk
        if thr < 0 then brakeInput = math.max(brakeInput, math.abs(thr)) end
        
        if brakeInput > 0 then
            local decel = TuningSettings.brakeForce * mass * brakeInput * 5
            if forwardSpeed > 1 then
                forceMag = -decel
            elseif forwardSpeed < -1 then
                forceMag = decel
            end
        end
        
        -- نطبق قوة الدفع على الاتجاه المسطح بس
        vf.Force = flatLook * forceMag
    end)
end

local function DetachCar()
    if AttachLoop then AttachLoop:Disconnect(); AttachLoop = nil end
    CurrentCar    = nil
    SmoothedRPM   = 0
    SmoothedSpeed = 0
    LiveData.rpm=0; LiveData.speed=0; LiveData.throttle=0
    LiveData.brake=0; LiveData.boost=0; LiveData.gear="N"
    UpdateGauges()
    SetStatus(TR("noVehicle"), Color3.fromRGB(255,100,30))
end

-- ATTACH
---------------------------------------------------------
local function AttachToCar(seat)
    if AttachLoop then AttachLoop:Disconnect(); AttachLoop = nil end

    local car = seat:FindFirstAncestorWhichIsA("Model")
        or (seat.Parent and seat.Parent:IsA("Model") and seat.Parent)
        or seat.Parent

    CurrentCar    = car
    LastSeatPart  = seat
    SmoothedRPM   = 0
    SmoothedSpeed = 0

    local name = (car and car.Name) or seat.Name
    if #name > 20 then name = name:sub(1,18).."..." end
    SetStatus(TR("ecu")..name, Color3.fromRGB(50,255,100))

    -- 1. المحاولة الأولى: للسيارات العادية
    if seat:IsA("VehicleSeat") then
        pcall(function()
            seat.MaxSpeed       = TuningSettings.maxSpeed
            seat.TorqueStrength = TuningSettings.engineTorque
            seat.BrakeForce     = TuningSettings.brakeForce
        end)
    end

    local lastTime = tick()
    AttachLoop = RunService.Heartbeat:Connect(function()
        if not seat or not seat.Parent then
            LastSeatPart = nil
            DetachCar()
            return
        end
        local now = tick()
        local dt  = math.min(now - lastTime, 0.1)
        lastTime  = now

        -- ✅ Force tuning كل frame
        ForceTuneFrame(seat, dt)

        ReadCarData(seat, dt)
        UpdateGauges()
    end)
end

-- AUTO ATTACH POLLING
---------------------------------------------------------
local lastScanTick = 0
local SCAN_GAP     = 0.3

RunService.Heartbeat:Connect(function()
    local char = LP.Character
    if not char then
        if LastSeatPart ~= nil then DetachCar(); LastSeatPart=nil end
        return
    end
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not hum then
        if LastSeatPart ~= nil then DetachCar(); LastSeatPart=nil end
        return
    end

    -- الطريقة 1: SeatPart مباشرة
    local sp = hum.SeatPart
    if sp and sp:IsA("VehicleSeat") then
        if sp ~= LastSeatPart then
            LastSeatPart = sp
            AttachToCar(sp)
        end
        return
    end

    -- الطريقة 2: Occupant scan كل 0.3 ثانية
    local now = tick()
    if now - lastScanTick < SCAN_GAP then return end
    lastScanTick = now

    local foundSeat = nil
    if sp and not sp:IsA("VehicleSeat") then
        local car = sp:FindFirstAncestorWhichIsA("Model")
        if car then
            local vs = car:FindFirstChildWhichIsA("VehicleSeat", true)
            foundSeat = vs
        end
    end
    if not foundSeat then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("VehicleSeat") and obj.Occupant == hum then
                foundSeat = obj; break
            end
        end
    end

    if foundSeat and foundSeat ~= LastSeatPart then
        LastSeatPart = foundSeat
        AttachToCar(foundSeat)
    elseif not foundSeat and LastSeatPart ~= nil then
        LastSeatPart = nil
        DetachCar()
    end
end)

local function ApplyLanguage()
    -- Status
    if CurrentCar == nil then
        StatusLabel.Text = TR("noVehicle")
    end
    -- Lang button
    LangBtn.Text = TR("langBtn")
    -- Tabs
    for _, ref in ipairs(TabLabelRefs) do
        ref.btn.Text = TR(ref.key)
    end
    -- Section labels
    for _, ref in ipairs(SectionLabels) do
        ref.lbl.Text = TR(ref.key)
    end
    -- Setting labels
    for _, ref in ipairs(SettingLabels) do
        ref.lbl.Text = TR(ref.key)
    end
    -- Buttons
    if ForceTuneBtn then
        ForceTuneBtn.Text = TuningActive and TR("tuningOn") or TR("forceTuning")
    end
    if ForceAttachBtn then
        ForceAttachBtn.Text = TR("forceAttach")
    end
end

end)
