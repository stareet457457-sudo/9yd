    local function smartUndergroundDrive(carModel, targetPos, speed)
        local primary = carModel.PrimaryPart or carModel:FindFirstChildWhichIsA("BasePart")
        if not primary then return end
        
        local expectedPos = carModel:GetPivot().Position
        local reached = false
        
        local connection
        connection = v15.Heartbeat:Connect(function(dt)
            if not carModel or not carModel.Parent then
                reached = true
                return
            end
            
            local direction = (targetPos - expectedPos)
            local distToTarget = direction.Magnitude
            local moveDist = speed * dt
            
            if distToTarget <= moveDist or distToTarget < 0.25 then
                expectedPos = targetPos
                local curLook = carModel:GetPivot().LookVector
                local flatLook = Vector3.new(curLook.X, 0, curLook.Z)
                if flatLook.Magnitude < 0.001 then flatLook = Vector3.new(0, 0, 1) end
                carModel:PivotTo(CFrame.lookAt(expectedPos, expectedPos + flatLook.Unit))
                if primary and primary.Parent then
                    primary.AssemblyLinearVelocity = Vector3.zero
                    primary.AssemblyAngularVelocity = Vector3.zero
                end
                reached = true
                return
            end
            
            direction = direction.Unit
            local flatDir = Vector3.new(direction.X, 0, direction.Z)
            if flatDir.Magnitude < 0.001 then flatDir = Vector3.new(0, 0, 1) end
            flatDir = flatDir.Unit
            
            expectedPos = expectedPos + (direction * moveDist)
            carModel:PivotTo(CFrame.lookAt(expectedPos, expectedPos + flatDir))
            if primary and primary.Parent then
                primary.AssemblyLinearVelocity = Vector3.zero
                primary.AssemblyAngularVelocity = Vector3.zero
            end
        end)

            task.wait(0.03)
        end
        if connection then
            connection:Disconnect()
        end
    end
